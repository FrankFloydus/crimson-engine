# PPQSort is header-only. Reproduce Tracy's two maintained upstream patches
# in a generated build copy, never in the verified source cache.
set(ppqsort_build_source "${CMAKE_CURRENT_BINARY_DIR}/dependencies/ppqsort-patched")
if(NOT EXISTS "${ppqsort_build_source}/.tracy-patches-applied-v2")
    file(MAKE_DIRECTORY "${ppqsort_build_source}")
    file(COPY "${CRIMSON_PPQSORT_SOURCE}/include" DESTINATION "${ppqsort_build_source}")
    # The pinned 1.0.6 header has an extra <new> include outside Tracy's
    # nodebug patch context. Apply the same two exact edits with checked
    # tokens, independently of a surrounding consumer Git worktree.
    set(ppq_parameters "${ppqsort_build_source}/include/ppqsort/parameters.h")
    file(READ "${ppq_parameters}" ppq_parameters_text)
    if(NOT ppq_parameters_text MATCHES "#ifndef NDEBUG")
        message(FATAL_ERROR "Pinned PPQSort debug guard differs from Tracy's patch.")
    endif()
    string(REPLACE "#ifndef NDEBUG" "// Tracy 0.14.1 ppqsort-nodebug.patch\n#if 0"
        ppq_parameters_text "${ppq_parameters_text}")
    file(WRITE "${ppq_parameters}" "${ppq_parameters_text}")
    set(ppq_thread_pool "${ppqsort_build_source}/include/ppqsort/parallel/cpp/thread_pool.h")
    file(READ "${ppq_thread_pool}" ppq_thread_pool_text)
    if(NOT ppq_thread_pool_text MATCHES "std::binary_semaphore threads_done_semaphore_")
        message(FATAL_ERROR "Pinned PPQSort semaphore differs from Tracy's patch.")
    endif()
    string(REPLACE "std::binary_semaphore threads_done_semaphore_"
        "std::counting_semaphore<> threads_done_semaphore_" ppq_thread_pool_text "${ppq_thread_pool_text}")
    file(WRITE "${ppq_thread_pool}" "${ppq_thread_pool_text}")
    foreach(patched_header "${ppq_parameters}" "${ppq_thread_pool}")
        if(NOT EXISTS "${patched_header}")
            message(FATAL_ERROR "Cannot prepare generated PPQSort dependency.")
        endif()
    endforeach()
    file(WRITE "${ppqsort_build_source}/.tracy-patches-applied-v2" "Tracy 0.14.1 / PPQSort 1.0.6\n")
endif()
add_library(CrimsonTracyPpqsort INTERFACE)
add_library(PPQSort::PPQSort ALIAS CrimsonTracyPpqsort)
target_include_directories(CrimsonTracyPpqsort INTERFACE "${ppqsort_build_source}/include")
target_link_libraries(CrimsonTracyPpqsort INTERFACE Threads::Threads)
target_compile_features(CrimsonTracyPpqsort INTERFACE cxx_std_20)

# Server-only Python binding. No TracyClient, application binding, DLL or
# second instrumentation runtime is created by this tools project.
find_package(Python 3.10 EXACT COMPONENTS Interpreter Development.Module REQUIRED)
set(PYBIND11_FINDPYTHON ON)
find_package(pybind11 2.13.6 EXACT CONFIG REQUIRED)
set(binding_patch "${CMAKE_CURRENT_LIST_DIR}/server-bindings-correctness.patch")
set(binding_original "${CRIMSON_TRACY_SOURCE}/python/bindings/ServerModule.cpp")
set(binding_generated_root "${CMAKE_CURRENT_BINARY_DIR}/generated/tracy-bindings")
set(binding_generated "${binding_generated_root}/python/bindings/ServerModule.cpp")
file(SHA256 "${binding_original}" binding_source_hash)
file(SHA256 "${binding_patch}" binding_patch_hash)
set(binding_identity "${binding_source_hash}/${binding_patch_hash}")
set(binding_stamp "${binding_generated_root}/.crimson-binding-patch")
set(binding_previous_identity "")
if(EXISTS "${binding_stamp}")
    file(READ "${binding_stamp}" binding_previous_identity)
endif()
if(NOT binding_previous_identity STREQUAL binding_identity OR NOT EXISTS "${binding_generated}")
    file(MAKE_DIRECTORY "${binding_generated_root}/python/bindings")
    configure_file("${binding_original}" "${binding_generated}" COPYONLY)
    find_package(Git REQUIRED)
    execute_process(COMMAND "${CMAKE_COMMAND}" -E env
        "GIT_CEILING_DIRECTORIES=${CMAKE_CURRENT_BINARY_DIR}/generated"
        "${GIT_EXECUTABLE}" apply --whitespace=nowarn "${binding_patch}"
        WORKING_DIRECTORY "${binding_generated_root}" RESULT_VARIABLE binding_patch_result)
    if(NOT binding_patch_result EQUAL 0)
        message(FATAL_ERROR "Cannot apply the pinned Tracy analysis binding correctness patch.")
    endif()
    file(WRITE "${binding_stamp}" "${binding_identity}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${binding_original}" "${binding_patch}")
pybind11_add_module(TracyServerBindings MODULE "${binding_generated}")
# Preserve the source file's relative server includes without modifying or
# copying the verified Tracy tree. Only this generated file is patched.
target_include_directories(TracyServerBindings PRIVATE "${CRIMSON_TRACY_SOURCE}/python/bindings")
target_link_libraries(TracyServerBindings PRIVATE TracyServer)
install(TARGETS tracy-capture tracy-csvexport TracyServerBindings RUNTIME DESTINATION . LIBRARY DESTINATION .)
