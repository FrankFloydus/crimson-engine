cmake_minimum_required(VERSION 3.25)
project(CrimsonDecalProof LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use Build.cmd Release decal-proof for the Windows x64 MSVC proof.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
get_filename_component(CC_DECAL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(Eigen3 CONFIG REQUIRED)
find_package(GeometryCentral CONFIG REQUIRED)
include("${CC_DECAL_ROOT}/cmake/Decals.cmake")
target_sources(crimson-decals PRIVATE "${CC_DECAL_ROOT}/src/assets/AtomicFile.cpp")
enable_testing()
add_executable(decal_surface_tests "${CC_DECAL_ROOT}/tests/decal_surface_tests.cpp")
target_link_libraries(decal_surface_tests PRIVATE crimson-decals)
target_compile_options(decal_surface_tests PRIVATE /W4 /permissive- /utf-8)
add_test(NAME decal_surface_core COMMAND decal_surface_tests)
add_custom_target(decal_proofs DEPENDS decal_surface_tests)
add_executable(decal_chart_dependency_tests "${CC_DECAL_ROOT}/tests/decal_chart_dependency_tests.cpp")
target_link_libraries(decal_chart_dependency_tests PRIVATE geometry-central::geometry-central)
add_dependencies(decal_proofs decal_chart_dependency_tests)
add_test(NAME decal_chart_dependency_core COMMAND decal_chart_dependency_tests)
include("${CMAKE_CURRENT_SOURCE_DIR}/HostProofs.cmake" OPTIONAL)
if(TARGET decal-proof-math)
    target_link_libraries(decal_surface_tests PRIVATE decal-proof-math)
    add_executable(decal_resolve_tests
        "${CC_DECAL_ROOT}/tests/decal_resolve_tests.cpp"
        "${CC_DECAL_ROOT}/src/decals/DecalModelSurface.cpp")
    target_link_libraries(decal_resolve_tests PRIVATE crimson-decals decal-proof-math)
    target_compile_options(decal_resolve_tests PRIVATE /W4 /permissive- /utf-8)
    add_test(NAME decal_resolve_core COMMAND decal_resolve_tests)
    add_dependencies(decal_proofs decal_resolve_tests)
    add_executable(decal_cache_tests "${CC_DECAL_ROOT}/tests/decal_cache_tests.cpp")
    target_link_libraries(decal_cache_tests PRIVATE crimson-decals decal-proof-math)
    target_compile_options(decal_cache_tests PRIVATE /W4 /permissive- /utf-8)
    add_test(NAME decal_cache_core COMMAND decal_cache_tests)
    add_dependencies(decal_proofs decal_cache_tests)
endif()
